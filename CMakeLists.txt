cmake_minimum_required(VERSION 3.16)
project(SupremeP0 VERSION 0.0.7 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use a Win32/x86 toolchain for Windows P0")
endif()
add_library(contracts INTERFACE)
target_include_directories(contracts INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
foreach(target supreme_diagnostic p0_tests p0_admission_tests)
 if(target STREQUAL "supreme_diagnostic")
  add_executable(${target} src/bootstrap/main.cpp)
 elseif(target STREQUAL "p0_tests")
  add_executable(${target} tests/test.cpp)
 else()
  add_executable(${target} tests/test_admission.cpp)
 endif()
 target_link_libraries(${target} PRIVATE contracts)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
enable_testing()
add_test(NAME P0_contracts_and_host COMMAND p0_tests)
add_test(NAME P0_offline_diagnostic COMMAND supreme_diagnostic)

add_test(NAME P0_admission COMMAND p0_admission_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME P0_preflight COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_preflight.py -v)
endif()

# P0 synthetic models: no native admission or production bootstrap dependency.
add_library(p0_session STATIC src/modules/session/model/session.cpp)
add_library(p0_submission STATIC src/modules/submission/model/draft.cpp)
foreach(target p0_session p0_submission)
 target_link_libraries(${target} PUBLIC contracts)
endforeach()
add_executable(p0_workflow_tests tests/test_workflow.cpp)
target_link_libraries(p0_workflow_tests PRIVATE p0_session p0_submission)
foreach(target p0_session p0_submission p0_workflow_tests)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
add_test(NAME P0_workflow_models COMMAND p0_workflow_tests)

# Synthetic C ABI only: not a DLL, not linked to the bootstrap.
add_library(p0_extension_mock STATIC src/engine_bridge/extension/mock.cpp)
target_link_libraries(p0_extension_mock PUBLIC contracts)
add_executable(p0_extension_tests tests/test_extension.c)
set_target_properties(p0_extension_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
target_link_libraries(p0_extension_tests PRIVATE p0_extension_mock)
foreach(target p0_extension_mock p0_extension_tests)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
add_test(NAME P0_extension_c_abi COMMAND p0_extension_tests)
if(Python3_Interpreter_FOUND)
 add_test(NAME P0_reference_inspection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference_inspection.py -v)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME P0_native_research COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_research.py -v)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME P0_baseline_snapshot COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_baseline_snapshot.py -v)
endif()

add_executable(p0_timing_tests tests/test_timing.cpp)
target_link_libraries(p0_timing_tests PRIVATE contracts)
if(MSVC)
 target_compile_options(p0_timing_tests PRIVATE /W4 /WX)
else()
 target_compile_options(p0_timing_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

 add_test(NAME P0_submission_preview COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_submission_preview.py -v)
